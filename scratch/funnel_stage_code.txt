149: /**
150:  * Envia as mensagens de um determinado estágio de funil configurado no banco de dados.
151:  */
152: export async function sendFunnelStage(
153:   db: D1Database,
154:   whatsappApiId: string,
155:   phone: string,
156:   automation: any,
157:   contact: any,
158:   state: any,
159:   stageKey: string,
160:   kv: any,
161:   extraVars: Record<string, string> = {},
162:   skipAwait = false
163: ): Promise<{ success: boolean; sent: boolean; messageLog: string[]; promises?: Promise<any>[] }> {
164:   try {
165:     const stage = await db.prepare(
166:       "SELECT id, enabled, rewrite_mode, variations FROM automation_funnel_stages WHERE automation_id = ? AND stage_key = ?"
167:     ).bind(automation.id, stageKey).first<{ id: string; enabled: number; rewrite_mode: string; variations: string }>();
168: 
169:     if (!stage || !stage.enabled) {
170:       console.log(`[Funnel Stage] ${stageKey} para ${phone} está desativado ou não configurado.`);
171:       return { success: true, sent: false, messageLog: [] };
172:     }
173: 
174:     const fieldsRes = await db.prepare(
175:       "SELECT * FROM automation_funnel_fields WHERE stage_id = ? ORDER BY sort_order ASC"
176:     ).bind(stage.id).all<{ type: string; content: string; file_name: string | null }>();
177:     const fields = fieldsRes.results || [];
178: 
179:     if (fields.length === 0) {
180:       console.log(`[Funnel Stage] Nenhuns campos encontrados para ${stageKey}`);
181:       return { success: true, sent: false, messageLog: [] };
182:     }
183: 
184:     const messageLog: string[] = [];
185:     const firstName = (contact.name || 'amiga').split(/\s+/)[0] || 'amiga';
186:     const attendantName = (automation as any).attendant_name || 'Julia';
187: 
188:     const replaceVariables = (txt: string) => {
189:       return txt
190:         .replace(/{{primeiro_nome}}/g, firstName)
191:         .replace(/{primeiro_nome}/g, firstName)
192:         .replace(/{{primeiro_name}}/g, firstName)
193:         .replace(/{primeiro_name}/g, firstName)
194:         .replace(/{{nome}}/g, contact.name || 'amiga')
195:         .replace(/{nome}/g, contact.name || 'amiga')
196:         .replace(/{{email_cliente}}/g, extraVars.email || state.client_email || '')
197:         .replace(/{email_cliente}/g, extraVars.email || state.client_email || '')
198:         .replace(/{{valor_pago}}/g, extraVars.valor_pago || (state.total_paid ? state.total_paid.toString() : '10.00'))
199:         .replace(/{valor_pago}/g, extraVars.valor_pago || (state.total_paid ? state.total_paid.toString() : '10.00'))
200:         .replace(/{{valor}}/g, extraVars.valor_pago || (state.total_paid ? state.total_paid.toString() : '10.00'))
201:         .replace(/{valor}/g, extraVars.valor_pago || (state.total_paid ? state.total_paid.toString() : '10.00'))
202:         .replace(/Julia/g, attendantName);
203:     };
204: 
205:     const promises: Promise<any>[] = [];
206:     let i = 0;
207:     while (i < fields.length) {
208:       const field = fields[i];
209: 
210:       // Sleep between blocks (except for the first one)
211:       if (i > 0) {
212:         await sleep(1500);
213:       }
214:       
215:       // Se for um documento/imagem/vídeo e o próximo também for, podemos agrupar todos os consecutivos
216:       if (field.type === 'document' || field.type === 'image' || field.type === 'video') {
217:         const batch = [field];
218:         let j = i + 1;
219:         while (j < fields.length && (fields[j].type === 'document' || fields[j].type === 'image' || fields[j].type === 'video')) {
220:           batch.push(fields[j]);
221:           j++;
222:         }
223:         
224:         console.log(`[Funnel Stage] Enviando lote de ${batch.length} mídias em paralelo para ${phone}`);
225:         const batchPromises = batch.map(f => {
226:           if (f.type === 'document') {
227:             return sendDocument(db, whatsappApiId, phone, f.content, f.file_name || 'apostila.pdf', kv, automation.id)
228:               .catch(err => console.error(`[Funnel Stage] Erro ao enviar documento ${f.file_name}:`, err));
229:           } else if (f.type === 'image') {
230:             return sendImage(db, whatsappApiId, phone, f.content, f.file_name || undefined, kv, automation.id)
231:               .catch(err => console.error(`[Funnel Stage] Erro ao enviar imagem:`, err));
232:           } else {
233:             return sendVideo(db, whatsappApiId, phone, f.content, f.file_name || undefined, kv, automation.id)
234:               .catch(err => console.error(`[Funnel Stage] Erro ao enviar vídeo:`, err));
235:           }
236:         });
237:         
238:         promises.push(...batchPromises);
239:         
240:         // Registrar as mensagens enviadas
241:         for (const f of batch) {
242:           if (f.type === 'document') {
243:             messageLog.push(`[PDF de receita enviado: ${f.file_name || 'documento'}]`);
244:           } else if (f.type === 'image') {
245:             messageLog.push(`[Imagem enviada]`);
246:           } else {
247:             messageLog.push(`[Vídeo enviado]`);
248:           }
249:         }
250:         
251:         // Avançar o ponteiro
252:         i = j;
253:       } else {
254:         // Envio sequencial para texto e áudio
255:         if (field.type === 'text') {
256:           let textToSend = field.content;
257:           if (stage.rewrite_mode === 'dynamic') {
258:             try {
259:               const dynList = await rewriteMessageViaLLM(db, automation.id, field.content, 1);
260:               textToSend = dynList[0] || field.content;
261:             } catch (err) {
262:               console.error("[Funnel Message Dynamic Rewrite] Erro:", err);
263:             }
264:           } else if (stage.rewrite_mode === 'static' && stage.variations) {
265:             try {
266:               const vars = JSON.parse(stage.variations);
267:               if (Array.isArray(vars) && vars.length > 0) {
268:                 const charSum = phone.split('').reduce((acc, char) => acc + char.charCodeAt(0), 0);
269:                 const idxVar = charSum % vars.length;
270:                 textToSend = vars[idxVar] || field.content;
271:               }
272:             } catch (e) {
273:               console.error("[Funnel Message Static Variations] Erro:", e);
274:             }
275:           }
276:           
277:           const formattedText = replaceVariables(textToSend);
278:           const p = sendText(db, whatsappApiId, phone, formattedText, kv, automation.id)
279:             .catch(err => console.error(`[Funnel Stage] Erro ao enviar texto:`, err));
280:           promises.push(p);
281:           messageLog.push(formattedText);
282:         } else if (field.type === 'audio') {
283:           const p = sendAudio(db, whatsappApiId, phone, field.content, kv, automation.id)
284:             .catch(err => console.error(`[Funnel Stage] Erro ao enviar áudio:`, err));
285:           promises.push(p);
286:           messageLog.push(`[Áudio de entrega enviado]`);
287:         }
288:         
289:         i++;
290:       }
291:     }
292: 
293:     if (skipAwait) {
294:       console.log(`[Funnel Stage] Pulando await interno de ${promises.length} disparos para ${phone}.`);
295:       return { success: true, sent: true, messageLog, promises };
296:     }
297: 
298:     console.log(`[Funnel Stage] Aguardando a conclusão de todos os ${promises.length} disparos...`);
299:     await Promise.all(promises);
300:     console.log(`[Funnel Stage] Todos os disparos concluídos para ${phone}.`);
301: 
302:     return { success: true, sent: true, messageLog };
303:   } catch (err) {
304:     console.error(`[Funnel Stage] Erro ao enviar estágio ${stageKey}:`, err);
305:     return { success: false, sent: false, messageLog: [] };
306:   }
307: }
