719:  */
720: export async function rewriteMessageViaLLM(
721:   db: D1Database,
722:   automationId: string,
723:   baseMessage: string,
724:   count = 1
725: ): Promise<string[]> {
726:   const systemPrompt = `Você é um redator de marketing altamente experiente. Sua tarefa é criar variações alternativas de uma mensagem de WhatsApp mantendo exatamente o mesmo objetivo, tom de voz, links e variáveis dinâmicas como {{nome}}, {{primeiro_nome}} e {{produto}}.
727: Retorne exatamente ${count} variações, uma em cada linha, no formato de JSON array de strings.

729: ["variação 1", "variação 2", "variação 3"]
730: 
731: Não adicione explicações, comentários ou formatação Markdown (como \`\`\`json). Retorne APENAS o JSON de array de strings puro e válido.`;
732: 
733:   const userPrompt = `Mensagem Base:\n"""\n${baseMessage}\n"""`;
734: 
735:   try {
736:     const response = await callLLM({
737:       db,
738:       automationId,
739:       systemPrompt,
740:       messages: [{ role: "user", content: userPrompt }]
741:     });
742: 
743:     const content = response.content.trim();
744:     const startIndex = content.indexOf('[');
745:     const endIndex = content.lastIndexOf(']');
746:     if (startIndex !== -1 && endIndex !== -1) {
747:       const jsonStr = content.substring(startIndex, endIndex + 1);
748:       const parsed = JSON.parse(jsonStr);
